document.addEventListener("DOMContentLoaded", function () {
  var controls = document.querySelectorAll("[data-place]");
  var field = document.querySelector(".jm-field");
  function show(id) {
    if (field) field.setAttribute("data-active", id);
    ["place-jhu", "place-penn"].forEach(function (p) {
      var panel = document.getElementById(p);
      if (panel) panel.hidden = p !== id;
    });
    controls.forEach(function (c) {
      var on = c.getAttribute("data-place") === id;
      c.setAttribute("aria-pressed", on ? "true" : "false");
      c.classList.toggle("is-active", on);
    });
  }
  controls.forEach(function (c) {
    c.addEventListener("click", function () { show(c.getAttribute("data-place")); });
    c.addEventListener("keydown", function (e) {
      if (c.tagName !== "BUTTON" && (e.key === "Enter" || e.key === " ")) { e.preventDefault(); show(c.getAttribute("data-place")); }
    });
  });
  show("place-jhu");
});
</script>
